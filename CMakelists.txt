cmake_minimum_required(VERSION 3.15)

project(DRBG C)

set(CMAKE_C_STANDARD 11)

include_directories(include drbgs crypto algorithms)

add_executable(DRBG
    src/main.c
    src/benchmark.c
    src/timer.c
    src/logger.c
    src/utils.c
    src/csv.c
    src/cli.c
    src/registry.c
    src/entropy.c

    crypto/crypto_hash.c
    crypto/crypto_hmac.c
    crypto/crypto_aes.c

    algorithms/hash_df.c
    algorithms/hash_math.c
    algorithms/hashgen.c

    drbgs/hash_drbg.c
    drbgs/dummy.c
)

add_executable(drbg_tests
    tests/test_main.c
    tests/hash_math_test.c
    tests/hash_df_test.c
    tests/hash_drbg_kat.c
    tests/hash_drbg_nist_kat.c
    tests/hash_drbg_nist_extended_kat.c
    tests/hash_drbg_boundary_test.c

    drbgs/hash_drbg.c

    algorithms/hash_df.c
    algorithms/hash_math.c
    algorithms/hashgen.c

    crypto/crypto_hash.c

    src/utils.c
)

target_include_directories(drbg_tests PRIVATE
    ${PROJECT_SOURCE_DIR}/include
    ${PROJECT_SOURCE_DIR}/drbgs
    ${PROJECT_SOURCE_DIR}/algorithms
    ${PROJECT_SOURCE_DIR}/crypto
)

target_link_libraries(drbg_tests PRIVATE
    OpenSSL::Crypto
    bcrypt
)
enable_testing()

add_test(
    NAME HashDRBGTests
    COMMAND drbg_tests
)
find_package(OpenSSL REQUIRED)

target_link_libraries(DRBG PRIVATE
    OpenSSL::Crypto
    bcrypt
)

add_executable(drbg_rsp_runner
    tests/nist_rsp_runner.c

    drbgs/hash_drbg.c

    algorithms/hash_df.c
    algorithms/hash_math.c
    algorithms/hashgen.c

    crypto/crypto_hash.c
)

target_include_directories(drbg_rsp_runner PRIVATE
    ${PROJECT_SOURCE_DIR}/include
    ${PROJECT_SOURCE_DIR}/drbgs
    ${PROJECT_SOURCE_DIR}/algorithms
    ${PROJECT_SOURCE_DIR}/crypto
)

target_link_libraries(drbg_rsp_runner PRIVATE
    OpenSSL::Crypto
)

set(NIST_DRBG_VECTOR_DIR "" CACHE PATH
    "Directory containing extracted NIST DRBG response files")

if(NIST_DRBG_VECTOR_DIR)
    file(GLOB_RECURSE NIST_HASH_DRBG_RESPONSE_FILES
        CONFIGURE_DEPENDS
        "${NIST_DRBG_VECTOR_DIR}/*.rsp"
    )


    list(FILTER NIST_HASH_DRBG_RESPONSE_FILES
        INCLUDE REGEX "[/\\\\]Hash_DRBG\\.rsp$")

    if(NIST_HASH_DRBG_RESPONSE_FILES)
        enable_testing()

        add_test(
            NAME NISTHashDRBGSHA256ResponseFiles
            COMMAND drbg_rsp_runner
                ${NIST_HASH_DRBG_RESPONSE_FILES}
        )
    else()
        message(WARNING
            "NIST_DRBG_VECTOR_DIR was set, but no Hash_DRBG.rsp files were found")
    endif()
endif()
