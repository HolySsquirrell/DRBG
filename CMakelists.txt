cmake_minimum_required(VERSION 3.15)

project(DRBG C)

set(CMAKE_C_STANDARD 11)

include_directories(include drbgs crypto)

add_executable(DRBG
    src/main.c
    src/benchmark.c
    src/timer.c
    src/logger.c
    src/utils.c
    src/csv.c
    src/cli.c
    src/registry.c
    src/entropy.c
    crypto/crypto_hash.c
    crypto/crypto_hmac.c
    crypto/crypto_aes.c
    drbgs/dummy.c
)
find_package(OpenSSL REQUIRED)

target_link_libraries(DRBG PRIVATE
    OpenSSL::Crypto
    bcrypt
)
