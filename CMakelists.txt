function(drbg_enable_diagnostics target_name)
    if(NOT TARGET ${target_name})
        message(
            FATAL_ERROR
            "Diagnostic target does not exist: ${target_name}"
        )
    endif()

    if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
        if(DRBG_ENABLE_STRICT_WARNINGS)
            target_compile_options(
                ${target_name}
                PRIVATE
                    -Wall
                    -Wextra
                    -Wpedantic
                    -Wformat=2
                    -Wshadow
                    -Wconversion
                    -Wsign-conversion
                    -Wstrict-prototypes
                    -Wundef
                    -Wcast-qual
            )
        endif()

        if(DRBG_ENABLE_ANALYZER)
            target_compile_options(
                ${target_name}
                PRIVATE
                    -fanalyzer
            )
        endif()

        if(DRBG_ENABLE_SANITIZERS)
            if(MINGW)
                message(
                    FATAL_ERROR
                    "ASan and UBSan runtimes are unavailable in the "
                    "current MSYS2 UCRT64 MinGW toolchain."
                )
            endif()

            target_compile_options(
                ${target_name}
                PRIVATE
                    -fsanitize=address,undefined
                    -fno-omit-frame-pointer
                    -fno-sanitize-recover=all
                    -O1
                    -g
            )

            target_link_options(
                ${target_name}
                PRIVATE
                    -fsanitize=address,undefined
            )
        endif()
    else()
        message(
            WARNING
            "Diagnostic settings currently support GCC only"
        )
    endif()
endfunction()

    NIST_DRBG_VECTOR_DIR
    ""
    CACHE PATH
    "Directory containing extracted NIST DRBG response files"
)

if(NIST_DRBG_VECTOR_DIR)
    file(
        GLOB_RECURSE
        NIST_DRBG_RESPONSE_FILES
        CONFIGURE_DEPENDS
        "${NIST_DRBG_VECTOR_DIR}/*.rsp"
    )

    set(
        NIST_HASH_DRBG_RESPONSE_FILES
        ${NIST_DRBG_RESPONSE_FILES}
    )

    list(
        FILTER
        NIST_HASH_DRBG_RESPONSE_FILES
        INCLUDE REGEX
        "[/\\\\]Hash_DRBG\\.rsp$"
    )

    if(NIST_HASH_DRBG_RESPONSE_FILES)
        add_test(
            NAME NISTHashDRBGSHA256ResponseFiles
            COMMAND
                drbg_rsp_runner
                ${NIST_HASH_DRBG_RESPONSE_FILES}
        )
    else()
        message(
            WARNING
            "NIST_DRBG_VECTOR_DIR was set, but no "
            "Hash_DRBG.rsp files were found"
        )
    endif()

    set(
        NIST_HMAC_DRBG_RESPONSE_FILES
        ${NIST_DRBG_RESPONSE_FILES}
    )

    list(
        FILTER
        NIST_HMAC_DRBG_RESPONSE_FILES
        INCLUDE REGEX
        "[/\\\\]HMAC_DRBG\\.rsp$"
    )

    if(NIST_HMAC_DRBG_RESPONSE_FILES)
        add_test(
            NAME NISTHMACDRBGSHA256ResponseFiles
            COMMAND
                hmac_drbg_rsp_runner
                ${NIST_HMAC_DRBG_RESPONSE_FILES}
        )
    else()
        message(
            WARNING
            "NIST_DRBG_VECTOR_DIR was set, but no "
            "HMAC_DRBG.rsp files were found"
        )
    endif()
endif()
